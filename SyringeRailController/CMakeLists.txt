cmake_minimum_required(VERSION 3.22)

# --- Cross-compilation settings MUST be before project() ---
set(CMAKE_SYSTEM_NAME Generic)
set(CMAKE_SYSTEM_PROCESSOR arm)

# Prevent CMake from trying to compile a test program right away
set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)

# Note: Adjust TOOLCHAIN_PREFIX if your arm-none-eabi-gcc path is different
set(TOOLCHAIN_PREFIX "arm-none-eabi-")
set(CMAKE_C_COMPILER "${TOOLCHAIN_PREFIX}gcc")
set(CMAKE_ASM_COMPILER "${TOOLCHAIN_PREFIX}gcc")
set(CMAKE_OBJCOPY "${TOOLCHAIN_PREFIX}objcopy")
set(CMAKE_SIZE "${TOOLCHAIN_PREFIX}size")

# --- Define Project Name ---
project(SyringeRailController C ASM)

# --- Standard settings ---
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

# --- Board MCU Flags (STM32F407VG for Discovery Board) ---
set(MCU_FLAGS "-mcpu=cortex-m4 -mthumb -mfpu=fpv4-sp-d16 -mfloat-abi=hard")

# Required Linker Script (CubeMX generates this)
set(LINKER_SCRIPT "${CMAKE_CURRENT_SOURCE_DIR}/STM32F407VGTX_FLASH.ld")
set(CMAKE_EXE_LINKER_FLAGS "${MCU_FLAGS} -T\"${LINKER_SCRIPT}\" --specs=nano.specs --specs=nosys.specs -Wl,-Map=${PROJECT_NAME}.map -Wl,--gc-sections")

set(CMAKE_C_FLAGS "${MCU_FLAGS} -Wall -Wextra -fdata-sections -ffunction-sections")
set(CMAKE_C_FLAGS_DEBUG "-Og -g3 -DDEBUG")
set(CMAKE_C_FLAGS_RELEASE "-O3 -DNDEBUG")

# --- Include Directories ---
# Core/Inc will be where app.h and stepper.h live along with main.h
include_directories(
    Core/Inc
    Drivers/STM32F4xx_HAL_Driver/Inc
    Drivers/STM32F4xx_HAL_Driver/Inc/Legacy
    Drivers/CMSIS/Device/ST/STM32F4xx/Include
    Drivers/CMSIS/Include
)

# --- Source Files ---
# Include all app sources, HAL drivers, and the generated main
file(GLOB_RECURSE PROJECT_SOURCES
    "Core/Src/*.c"
)

# HAL Source Files
file(GLOB_RECURSE HAL_SOURCES
    "Drivers/STM32F4xx_HAL_Driver/Src/*.c"
)
list(FILTER HAL_SOURCES EXCLUDE REGEX ".*_template\\.c$")

# Auto-detect the startup file generated by CubeMX
file(GLOB STARTUP_FILE "startup_stm32f407*.s")

add_executable(${PROJECT_NAME}.elf 
    ${PROJECT_SOURCES} 
    ${HAL_SOURCES}
    ${STARTUP_FILE}
)

# --- Target Definitions ---
# Tells HAL to use the right MCU defines
target_compile_definitions(${PROJECT_NAME}.elf PRIVATE
    USE_HAL_DRIVER
    STM32F407xx
)

# --- Generate .bin and .hex for Flashing ---
add_custom_command(TARGET ${PROJECT_NAME}.elf POST_BUILD
    COMMAND ${CMAKE_OBJCOPY} -O binary ${PROJECT_NAME}.elf ${PROJECT_NAME}.bin
    COMMAND ${CMAKE_OBJCOPY} -O ihex ${PROJECT_NAME}.elf ${PROJECT_NAME}.hex
    COMMAND ${CMAKE_SIZE} ${PROJECT_NAME}.elf
)

# --- Flash Target Setup (OpenOCD) ---
add_custom_target(flash
    COMMAND openocd -f interface/stlink.cfg -f target/stm32f4x.cfg -c "program ${PROJECT_NAME}.bin 0x08000000 verify reset exit"
    DEPENDS ${PROJECT_NAME}.elf
)
